package com.networknt.schema;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ArrayNode;
import org.junit.Before;
import org.junit.Test;

import java.net.URI;
import java.util.Collections;
import java.util.Set;

import static org.junit.Assert.*;

/**
 * JUnit‑4 test suite for {@link OneOfValidator}.
 *
 * The tests create real {@link JsonSchema} objects through the public
 * {@link ValidationContext#newSchema(String, JsonNode, JsonSchema)} factory
 * and use a real {@link CollectorContext} together with a {@link ValidatorState}
 * (which is required by the validator implementation).  No mocking frameworks
 * are used – only the public constructors and methods that are part of the
 * library API.
 */
public class OneOfValidatorTest {

    private static final ObjectMapper MAPPER = new ObjectMapper();

    private ValidationContext validationContext;
    private CollectorContext collectorContext;
    private ExecutionContext executionContext;
    private ValidatorState validatorState;

    @Before
    public void setUp() {
        // -----------------------------------------------------------------
        // Build the minimal objects required by the library.
        // -----------------------------------------------------------------
        URIFactory uriFactory = new URIFactory();
        URNFactory urnFactory = new URNFactory();
        JsonMetaSchema metaSchema = JsonMetaSchema.getInstance(); // default meta‑schema
        JsonSchemaFactory schemaFactory = JsonSchemaFactory.getInstance();
        SchemaValidatorsConfig config = new SchemaValidatorsConfig();

        validationContext = new ValidationContext(uriFactory, urnFactory, metaSchema,
                schemaFactory, config);

        // CollectorContext is needed for the validator state handling.
        collectorContext = new CollectorContext();
        validatorState = new ValidatorState(); // default state (all flags false)
        collectorContext.add(ValidatorState.VALIDATOR_STATE_KEY, validatorState);

        executionContext = new ExecutionContext(collectorContext);
    }

    /**
     * Helper that builds a {@link OneOfValidator} from an array of subschema
     * definitions.
     *
     * @param subschemas array node containing the subschemas
     * @return a ready‑to‑use {@link OneOfValidator}
     */
    private OneOfValidator buildValidator(ArrayNode subschemas) {
        // The parent schema is not needed for these tests – we pass {@code null}.
        return new OneOfValidator("#/oneOf", subschemas, null, validationContext);
    }

    /**
     * Test the happy‑path where exactly one subschema matches.
     * The node is a JSON string; the subschemas are “type:string” and
     * “type:integer”.  Only the first subschema matches, therefore the validator
     * must return an empty error set.
     */
    @Test
    public void testValidate_ExactlyOneValidSchema() throws Exception {
        // -----------------------------------------------------------------
        // Build the OneOf schema: [ {"type":"string"}, {"type":"integer"} ]
        // -----------------------------------------------------------------
        ArrayNode oneOfArray = MAPPER.createArrayNode();
        oneOfArray.add(MAPPER.readTree("{\"type\":\"string\"}"));
        oneOfArray.add(MAPPER.readTree("{\"type\":\"integer\"}"));

        OneOfValidator validator = buildValidator(oneOfArray);

        // The instance to validate – a JSON string.
        JsonNode instance = MAPPER.readTree("\"hello\"");

        Set<ValidationMessage> errors = validator.validate(executionContext,
                instance, instance, "#");

        // No errors expected and the validator state must indicate a match.
        assertTrue("Expected no validation errors", errors.isEmpty());
        assertTrue("ValidatorState should have matchedNode == true",
                validatorState.isMatchedNode());
    }

    /**
     * Test the situation where **no** subschema validates.
     * The instance is a boolean; the subschemas accept only strings and integers.
     * The validator must return a OneOf error (code “oneOf”) and also expose the
     * child errors generated by each subschema.
     */
    @Test
    public void testValidate_NoValidSchema() throws Exception {
        // -----------------------------------------------------------------
        // OneOf schema: [ {"type":"string"}, {"type":"integer"} ]
        // -----------------------------------------------------------------
        ArrayNode oneOfArray = MAPPER.createArrayNode();
        oneOfArray.add(MAPPER.readTree("{\"type\":\"string\"}"));
        oneOfArray.add(MAPPER.readTree("{\"type\":\"integer\"}"));

        OneOfValidator validator = buildValidator(oneOfArray);

        // Instance that matches none of the subschemas.
        JsonNode instance = MAPPER.readTree("true");

        Set<ValidationMessage> errors = validator.validate(executionContext,
                instance, instance, "#");

        // Must contain the OneOf error.
        assertFalse("Expected validation errors", errors.isEmpty());

        // The first error must be the OneOf error (its code is “oneOf”).
        ValidationMessage first = errors.iterator().next();
        assertEquals("oneOf", first.getCode());

        // The message should contain the number of valid schemas (0).
        assertTrue("Message should contain '0' valid schemas",
                first.getMessage().contains("0"));

        // Child errors from each subschema must also be present.
        // There are exactly two subschemas, therefore we expect at least two
        // additional messages (type‑mismatch for string and integer).
        assertTrue("Expected at least three messages (1 OneOf + 2 child errors)",
                errors.size() >= 3);
    }

    /**
     * Test the case where **more than one** subschema validates.
     * The instance is the integer 5.  The subschemas are:
     *   1. {"type":"integer"}
     *   2. {"enum":[5]}
     *
     * Both subschemas accept the value, so the validator must report a OneOf
     * error (the number of valid schemas is 2) and must not report any child
     * errors because validation short‑circuits after the second match.
     */
    @Test
    public void testValidate_MultipleValidSchemas() throws Exception {
        // -----------------------------------------------------------------
        // OneOf schema: [ {"type":"integer"}, {"enum":[5]} ]
        // -----------------------------------------------------------------
        ArrayNode oneOfArray = MAPPER.createArrayNode();
        oneOfArray.add(MAPPER.readTree("{\"type\":\"integer\"}"));
        oneOfArray.add(MAPPER.readTree("{\"enum\":[5]}"));

        OneOfValidator validator = buildValidator(oneOfArray);

        JsonNode instance = MAPPER.readTree("5");

        Set<ValidationMessage> errors = validator.validate(executionContext,
                instance, instance, "#");

        // The validator must return a OneOf error (code “oneOf”) and the count
        // must be “2”.
        assertFalse("Expected validation errors", errors.isEmpty());
        ValidationMessage first = errors.iterator().next();
        assertEquals("oneOf", first.getCode());
        assertTrue("Message should contain '2' valid schemas",
                first.getMessage().contains("2"));
    }

    /**
     * Verify the {@link OneOfValidator#walk(ExecutionContext, JsonNode, JsonNode, String, boolean)}
     * behaviour.
     *
     * When {@code shouldValidateSchema} is {@code false} the validator must *not*
     * invoke its own {@code validate} method – it simply walks each child schema.
     * Because the child schemas are walked with validation disabled, the returned
     * set must be empty even if the instance would otherwise be invalid.
     */
    @Test
    public void testWalk_WithValidationDisabled() throws Exception {
        // -----------------------------------------------------------------
        // OneOf schema: [ {"type":"string"}, {"type":"integer"} ]
        // -----------------------------------------------------------------
        ArrayNode oneOfArray = MAPPER.createArrayNode();
        oneOfArray.add(MAPPER.readTree("{\"type\":\"string\"}"));
        oneOfArray.add(MAPPER.readTree("{\"type\":\"integer\"}"));

        OneOfValidator validator = buildValidator(oneOfArray);

        // Instance that would be invalid for the first subschema (integer) and
        // valid for the second (string) – we use a boolean to guarantee *no*
        // match.
        JsonNode instance = MAPPER.readTree("false");

        // Walk with validation disabled.
        Set<ValidationMessage> messages = validator.walk(executionContext,
                instance, instance, "#", false);

        // No messages must be produced.
        assertTrue("walk with shouldValidateSchema=false must return empty set",
                messages.isEmpty());
    }

    /**
     * Verify that {@link OneOfValidator#preloadJsonSchema()} simply forwards the
     * call to each child schema.  The method does not return a value, so the
     * test only checks that it executes without throwing an exception.
     */
    @Test
    public void testPreloadJsonSchemaExecutesWithoutError() {
        // -----------------------------------------------------------------
        // OneOf schema: [ {"type":"string"}, {"type":"integer"} ]
        // -----------------------------------------------------------------
        ArrayNode oneOfArray = MAPPER.createArrayNode();
        oneOfArray.add(MAPPER.readTree("{\"type\":\"string\"}"));
        oneOfArray.add(MAPPER.readTree("{\"type\":\"integer\"}"));

        OneOfValidator validator = buildValidator(oneOfArray);

        // The call should complete without throwing.
        validator.preloadJsonSchema();
    }
}
